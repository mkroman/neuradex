# CI tool manifest.
#
# Each [tool] block is read by install-tools.sh. Keep repo, version and
# sha256 contiguous: the Renovate manager in renovate.json matches that block
# head. All other keys are installer details.
#
# The cargo tools baked into the Docker build (cargo-auditable, cargo-chef)
# live in docker-tools.txt instead, so changing them does not invalidate the
# CI tool layer.
#
#   [tool]
#   repo=owner/repository
#   version=v1.2.3
#   sha256=<64 hex characters>
#   tag=${version}
#   asset=<release asset template>
#   binary=<path inside the archive>
#   archive=tar.gz|tar.xz|raw
#
# version is the upstream release tag as published (with or without a leading
# "v"). Renovate updates it and sha256 with the github-release-attachments
# datasource, which requests releases by their exact tag.
#
# ${version} expands to the full tag. ${version_nov} expands to the tag with a
# leading "v" removed, for release asset names that omit the prefix.
#
